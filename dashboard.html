/* ===================================== TEACHER MANAGEMENT ===================================== */ let editingTeacherId = null; /* LOAD TEACHERS */ async function loadTeachers(searchText = "") { const table = document.getElementById("teacherTable"); table.innerHTML = ` Loading teachers... `; const { data, error } = await supabaseClient .from("teachers") .select("*") .order("id", { ascending:false }); if(error){ console.error(error); table.innerHTML = ` Failed to load teachers. `; return; } let teachers = data || []; const search = searchText.toLowerCase().trim(); if(search){ teachers = teachers.filter(function(teacher){ const text = ( teacher.full_name + " " + teacher.username + " " + (teacher.class_name || "") + " " + (teacher.phone || "") ).toLowerCase(); return text.includes(search); }); } if(teachers.length === 0){ table.innerHTML = ` No teachers registered yet. `; return; } teachers.forEach(function(teacher,index){ const row = document.createElement("tr"); row.innerHTML = ` ${index + 1} ${escapeHTML(teacher.full_name)} ${escapeHTML(teacher.username)} ${escapeHTML(teacher.class_name || "-")} ${escapeHTML(teacher.phone || "-")} `; table.appendChild(row); }); } /* REGISTER / UPDATE TEACHER */ document .getElementById("teacherForm") .addEventListener( "submit", async function(event){ event.preventDefault(); const message = document.getElementById( "teacherMessage" ); const fullName = document .getElementById("teacherName") .value .trim(); const username = document .getElementById("teacherUsername") .value .trim(); const className = document .getElementById("teacherClass") .value; const phone = document .getElementById("teacherPhone") .value .trim(); if(editingTeacherId !== null){ const { error } = await supabaseClient .from("teachers") .update({ full_name: fullName, username: username, class_name: className || null, phone: phone || null }) .eq("id",editingTeacherId); if(error){ message.style.color = "red"; message.textContent = "Failed to update teacher: " + error.message; return; } message.style.color = "green"; message.textContent = "Teacher updated successfully."; editingTeacherId = null; document .getElementById("teacherSubmit") .textContent = "➕ REGISTER TEACHER"; document .getElementById("teacherForm") .reset(); loadTeachers(); return; } const { error } = await supabaseClient .from("teachers") .insert({ full_name: fullName, username: username, class_name: className || null, phone: phone || null }); if(error){ console.error(error); message.style.color = "red"; if(error.code === "23505"){ message.textContent = "Username already exists."; }else{ message.textContent = "Failed to register teacher: " + error.message; } return; } message.style.color = "green"; message.textContent = "Teacher registered successfully."; document .getElementById("teacherForm") .reset(); loadTeachers(); } ); /* EDIT TEACHER */ async function editTeacher(id){ const { data, error } = await supabaseClient .from("teachers") .select("*") .eq("id",id) .single(); if(error){ alert( "Failed to load teacher." ); return; } document .getElementById("teacherName") .value = data.full_name; document .getElementById("teacherUsername") .value = data.username; document .getElementById("teacherClass") .value = data.class_name || ""; document .getElementById("teacherPhone") .value = data.phone || ""; editingTeacherId = id; document .getElementById("teacherSubmit") .textContent = "💾 UPDATE TEACHER"; window.scrollTo({ top:0, behavior:"smooth" }); } /* DELETE TEACHER */ async function deleteTeacher(id){ const confirmDelete = confirm( "Are you sure you want to delete this teacher?" ); if(!confirmDelete){ return; } const { error } = await supabaseClient .from("teachers") .delete() .eq("id",id); if(error){ alert( "Failed to delete teacher: " + error.message ); return; } loadTeachers(); } /* SEARCH TEACHERS */ document .getElementById("teacherSearch") .addEventListener( "input", function(){ loadTeachers( this.value );
